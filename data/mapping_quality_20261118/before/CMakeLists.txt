cmake_minimum_required(VERSION 3.8)
project(robot_slam)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
foreach(dep robot_interfaces diagnostic_msgs geometry_msgs map_msgs message_filters
    nav_msgs nav2_map_server pcl_conversions rclcpp rcl_interfaces sensor_msgs
    std_srvs tf2 tf2_geometry_msgs tf2_ros)
  find_package(${dep} REQUIRED)
endforeach()
find_package(Eigen3 REQUIRED)
find_package(GTSAM REQUIRED)
find_package(Open3D REQUIRED)
find_package(PCL REQUIRED COMPONENTS common filters io segmentation)
set(BUILD_HELPER ON CACHE BOOL "small_gicp helper" FORCE)
set(BUILD_TESTS OFF CACHE BOOL "small_gicp tests" FORCE)
set(BUILD_EXAMPLES OFF CACHE BOOL "small_gicp examples" FORCE)
set(BUILD_BENCHMARKS OFF CACHE BOOL "small_gicp benchmarks" FORCE)
set(BUILD_PYTHON_BINDINGS OFF CACHE BOOL "small_gicp python" FORCE)
add_subdirectory(../3rdparty/small_gicp ${CMAKE_CURRENT_BINARY_DIR}/small_gicp EXCLUDE_FROM_ALL)
include_directories(include ${PCL_INCLUDE_DIRS})
add_library(gravity_alignment_core STATIC src/gravity_alignment.cpp)
target_link_libraries(gravity_alignment_core Eigen3::Eigen)
add_executable(odometry_frame_converter src/odometry_frame_converter.cpp)
ament_target_dependencies(odometry_frame_converter nav_msgs rclcpp sensor_msgs tf2 tf2_geometry_msgs tf2_ros)
target_link_libraries(odometry_frame_converter gravity_alignment_core Eigen3::Eigen)
add_library(localization_pose_cache STATIC src/localization_pose_cache.cpp)
target_link_libraries(localization_pose_cache Eigen3::Eigen)
add_executable(open3d_localizer src/open3d_localizer.cpp)
ament_target_dependencies(open3d_localizer robot_interfaces geometry_msgs message_filters
  nav_msgs pcl_conversions rclcpp rcl_interfaces sensor_msgs tf2 tf2_ros)
target_link_libraries(open3d_localizer localization_pose_cache ${PCL_LIBRARIES} Open3D::Open3D)
add_executable(mapping_tf_bridge src/mapping_tf_bridge.cpp)
ament_target_dependencies(mapping_tf_bridge message_filters nav_msgs rclcpp tf2 tf2_geometry_msgs tf2_ros)
target_link_libraries(mapping_tf_bridge Eigen3::Eigen)
add_library(loop_closure_core STATIC src/colored_map_builder.cpp src/descriptor_index.cpp
  src/loop_registration.cpp src/map_rebuilder.cpp src/point_cloud_utils.cpp
  src/pose_graph.cpp src/solid_descriptor.cpp)
ament_target_dependencies(loop_closure_core map_msgs nav_msgs rclcpp)
target_link_libraries(loop_closure_core ${PCL_LIBRARIES} gtsam small_gicp)
add_executable(loop_closure_mapper src/loop_closure_mapper.cpp)
ament_target_dependencies(loop_closure_mapper diagnostic_msgs robot_interfaces map_msgs
  message_filters nav_msgs nav2_map_server pcl_conversions rclcpp sensor_msgs)
target_link_libraries(loop_closure_mapper loop_closure_core)
install(TARGETS small_gicp LIBRARY DESTINATION lib)
install(TARGETS odometry_frame_converter open3d_localizer mapping_tf_bridge
  loop_closure_mapper DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config rviz DESTINATION share/${PROJECT_NAME})
install(PROGRAMS scripts/navigation_gate.py DESTINATION lib/${PROJECT_NAME} RENAME navigation_gate)
install(PROGRAMS scripts/save_map.py DESTINATION lib/${PROJECT_NAME} RENAME save_map)
ament_package()
