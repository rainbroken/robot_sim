cmake_minimum_required(VERSION 3.8)
project(robot_sim)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(gazebo REQUIRED)
find_package(gazebo_ros REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(tf2_geometry_msgs REQUIRED)
# Keep the signed encoder-velocity fix local; never replace the system plugin.
add_library(robot_sim_diff_drive SHARED src/gazebo_ros_diff_drive.cpp)
target_compile_features(robot_sim_diff_drive PUBLIC cxx_std_17)
target_include_directories(robot_sim_diff_drive SYSTEM PRIVATE ${GAZEBO_INCLUDE_DIRS})
target_link_directories(robot_sim_diff_drive PRIVATE ${GAZEBO_LIBRARY_DIRS})
target_link_libraries(robot_sim_diff_drive ${GAZEBO_LIBRARIES})
ament_target_dependencies(robot_sim_diff_drive gazebo_ros rclcpp geometry_msgs nav_msgs
                          tf2_ros tf2_geometry_msgs)
install(TARGETS robot_sim_diff_drive LIBRARY DESTINATION lib)
ament_export_libraries(robot_sim_diff_drive)
add_library(robot_sim_random_pedestrians SHARED src/random_pedestrians.cpp)
target_compile_features(robot_sim_random_pedestrians PUBLIC cxx_std_17)
target_compile_options(robot_sim_random_pedestrians PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(robot_sim_random_pedestrians SYSTEM PRIVATE ${GAZEBO_INCLUDE_DIRS})
target_link_directories(robot_sim_random_pedestrians PRIVATE ${GAZEBO_LIBRARY_DIRS})
target_link_libraries(robot_sim_random_pedestrians ${GAZEBO_LIBRARIES})
install(TARGETS robot_sim_random_pedestrians LIBRARY DESTINATION lib)
add_executable(generate_family_house src/generate_family_house.cpp)
target_compile_features(generate_family_house PUBLIC cxx_std_17)
target_compile_options(generate_family_house PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS generate_family_house DESTINATION lib/${PROJECT_NAME})
add_executable(mapping_route src/mapping_route.cpp)
target_compile_features(mapping_route PUBLIC cxx_std_17)
target_compile_options(mapping_route PRIVATE -Wall -Wextra -Wpedantic)
ament_target_dependencies(mapping_route rclcpp geometry_msgs tf2_ros sensor_msgs)
install(TARGETS mapping_route DESTINATION lib/${PROJECT_NAME})
add_executable(command_watchdog src/command_watchdog.cpp)
target_compile_features(command_watchdog PUBLIC cxx_std_17)
target_compile_options(command_watchdog PRIVATE -Wall -Wextra -Wpedantic)
ament_target_dependencies(command_watchdog rclcpp geometry_msgs std_msgs)
install(TARGETS command_watchdog DESTINATION lib/${PROJECT_NAME})
add_executable(cloud_to_scan src/cloud_to_scan.cpp)
target_compile_features(cloud_to_scan PUBLIC cxx_std_17)
target_compile_options(cloud_to_scan PRIVATE -Wall -Wextra -Wpedantic)
ament_target_dependencies(cloud_to_scan rclcpp sensor_msgs tf2_ros tf2_geometry_msgs)
install(TARGETS cloud_to_scan DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch worlds config rviz DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_pedestrian_clearance test/test_pedestrian_clearance.cpp)
  if(TARGET test_pedestrian_clearance)
    target_include_directories(test_pedestrian_clearance PRIVATE src)
  endif()
endif()
ament_package()
